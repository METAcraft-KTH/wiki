{{- /* Renders an image from assets/images. Args: src, width (display px), alt */ -}} {{- $w := int (.width | default 250) -}} {{- with resources.Get (path.Join "images" .src) -}} {{- $img := . -}} {{- $raster := ne .MediaType.SubType "svg" -}} {{- /* 2x width for hi-dpi screens; originals stay linked */ -}} {{- if and $raster (gt .Width (mul $w 2)) }}{{ $img = .Resize (printf "%dx" (mul $w 2)) }}{{ end -}} {{- else -}} {{- warnf "image %q not found in assets/images" .src -}} File:{{ .src }} {{- end -}}
